```sql
WITH evaluated AS (
    SELECT
        o.id AS opportunity_id,
        o.track,
        o.work_mode,
        o.remote_scope,
        o.location_country,
        o.title_family,
        m.qualification_decision,
        m.fit_score,
        m.hard_failure_code
    FROM public.opportunities AS o
    LEFT JOIN public.match_evaluations AS m ON m.opportunity_id = o.id
),
location_counts AS (
    SELECT
        COALESCE(NULLIF(BTRIM(location_country), ''), 'unknown') AS category,
        COUNT(*)::bigint AS n,
        ROW_NUMBER() OVER (ORDER BY COUNT(*) DESC, COALESCE(NULLIF(BTRIM(location_country), ''), 'unknown')) AS position
    FROM evaluated
    WHERE track = 'employment'
    GROUP BY COALESCE(NULLIF(BTRIM(location_country), ''), 'unknown')
),
metrics AS (
    SELECT 'track'::text AS metric, COALESCE(track, '<null>')::text AS category, COUNT(*)::bigint AS n, '{}'::jsonb AS value
    FROM public.opportunities GROUP BY track
    UNION ALL
    SELECT 'employment_decision', COALESCE(qualification_decision, '<null>'), COUNT(*)::bigint, '{}'::jsonb
    FROM evaluated WHERE track = 'employment' GROUP BY qualification_decision
    UNION ALL
    SELECT 'employment_hard_failure', COALESCE(hard_failure_code, '<null>'), COUNT(*)::bigint, '{}'::jsonb
    FROM evaluated WHERE track = 'employment' AND hard_failure_code IS NOT NULL GROUP BY hard_failure_code
    UNION ALL
    SELECT 'employment_fit_band',
           CASE WHEN fit_score IS NULL THEN 'missing'
                WHEN fit_score < 20 THEN '00-19'
                WHEN fit_score < 40 THEN '20-39'
                WHEN fit_score < 60 THEN '40-59'
                WHEN fit_score < 80 THEN '60-79'
                ELSE '80-100+' END,
           COUNT(*)::bigint, '{}'::jsonb
    FROM evaluated WHERE track = 'employment' GROUP BY 2
    UNION ALL
    SELECT 'employment_work_mode', COALESCE(work_mode, '<null>'), COUNT(*)::bigint, '{}'::jsonb
    FROM evaluated WHERE track = 'employment' GROUP BY work_mode
    UNION ALL
    SELECT 'employment_remote_scope', COALESCE(remote_scope, '<null>'), COUNT(*)::bigint, '{}'::jsonb
    FROM evaluated WHERE track = 'employment' GROUP BY remote_scope
    UNION ALL
    SELECT 'employment_location_country_top20', category, n, '{}'::jsonb FROM location_counts WHERE position <= 20
    UNION ALL
    SELECT 'employment_title_family_persistence',
           CASE WHEN NULLIF(BTRIM(title_family), '') IS NULL THEN 'missing' ELSE 'populated' END,
           COUNT(*)::bigint, '{}'::jsonb
    FROM evaluated WHERE track = 'employment'
    GROUP BY CASE WHEN NULLIF(BTRIM(title_family), '') IS NULL THEN 'missing' ELSE 'populated' END
    UNION ALL
    SELECT 'employment_fit_summary', 'all_scored', COUNT(fit_score)::bigint,
           jsonb_build_object(
               'minimum', MIN(fit_score),
               'median', PERCENTILE_CONT(0.50) WITHIN GROUP (ORDER BY fit_score),
               'p90', PERCENTILE_CONT(0.90) WITHIN GROUP (ORDER BY fit_score),
               'maximum', MAX(fit_score)
           )
    FROM evaluated WHERE track = 'employment' AND fit_score IS NOT NULL
    UNION ALL
    SELECT 'employment_projection_rows', 'all', COUNT(*)::bigint, '{}'::jsonb
    FROM public.feed_projection WHERE track = 'employment'
    UNION ALL
    SELECT 'employment_projection_title_family',
           CASE WHEN NULLIF(BTRIM(title_family), '') IS NULL THEN 'missing' ELSE 'populated' END,
           COUNT(*)::bigint, '{}'::jsonb
    FROM public.feed_projection WHERE track = 'employment'
    GROUP BY CASE WHEN NULLIF(BTRIM(title_family), '') IS NULL THEN 'missing' ELSE 'populated' END
    UNION ALL
    SELECT 'tracker_state', COALESCE(state, '<null>'), COUNT(*)::bigint, '{}'::jsonb
    FROM public.founder_triage_states GROUP BY state
    UNION ALL
    SELECT 'capture_time_utc', 'snapshot', NULL::bigint,
           jsonb_build_object('captured_at', TO_CHAR(CURRENT_TIMESTAMP AT TIME ZONE 'UTC', 'YYYY-MM-DD"T"HH24:MI:SS"Z"'))
)
SELECT metric, category, n, value
FROM metrics
ORDER BY metric, category;
```
