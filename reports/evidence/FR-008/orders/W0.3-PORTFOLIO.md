Metadata-only inline inventory command (run from the repository root):

```powershell
@'
import hashlib
import json
import os
from collections import Counter
from pathlib import Path

import yaml
from pypdf import PdfReader

root = Path(os.environ["FR008_PORTFOLIO_DIR"])
catalog = yaml.safe_load(Path("founder/cv_portfolio.yaml").read_text(encoding="utf-8"))
locked = {
    str(item["sha256"]).strip().lower()
    for item in catalog.get("variants", [])
    if item.get("status") == "final_locked" and item.get("sha256")
}
files = list(root.rglob("*.pdf"))
readable = 0
matches = 0
page_counts = Counter()
for path in files:
    try:
        digest = hashlib.sha256(path.read_bytes()).hexdigest()
        reader = PdfReader(str(path), strict=True)
        if reader.is_encrypted:
            continue
        page_counts[len(reader.pages)] += 1
        readable += 1
        matches += digest.lower() in locked
    except Exception:
        continue
print(json.dumps({
    "candidate_pdf_count": len(files),
    "readable_pdf_count": readable,
    "unreadable_pdf_count": len(files) - readable,
    "page_count_distribution": dict(sorted(page_counts.items())),
    "locked_catalog_variant_count": len(locked),
    "locked_hash_matches": matches,
    "unmatched_candidate_count": len(files) - matches,
}, sort_keys=True))
'@ | python -
```
